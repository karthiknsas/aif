from langchain_openai import ChatOpenAI
from foundry_local import FoundryLocalManager
import json
import re

manager = FoundryLocalManager("phi-3.5-mini")
llm = ChatOpenAI(base_url=manager.endpoint, api_key=manager.api_key)

# Router LLM decides
def llm_router(query: str):
    response = llm.invoke(f"""Decide mode only:
    Query: {query}
    Return: {{"mode": "agent" or "chat"}}
    """)

    match = re.search(r'"mode":\s*"(\w+)"', response.content)
    return match.group(1) if match else "chat"

# Test
query1 = "Search for Python books"
mode1 = llm_router(query1)
print(f"Query: {query1}")
print(f"LLM Decision: {mode1}")  # Output: agent ✅
